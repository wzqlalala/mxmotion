#-----------------------------------------------------------------------------
# 头文件搜索路径
#-----------------------------------------------------------------------------
include_directories(${CMAKE_CURRENT_SOURCE_DIR})
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../MainWindow)

#-----------------------------------------------------------------------------
# 自动添加include目录
#-----------------------------------------------------------------------------
set(CMAKE_INCLUDE_CURRENT_DIR ON)

#-----------------------------------------------------------------------------
# 源码扫描
#-----------------------------------------------------------------------------
file(GLOB _ui "*.ui")
file(GLOB _header "*.h")
file(GLOB _source "*.cpp")
qt5_wrap_ui(_interface ${_ui})

#-----------------------------------------------------------------------------
# 添加动态库目标
#-----------------------------------------------------------------------------
add_library(APIServer
	${_resource}
	${_interface}
	${_header}
	${_source}
)

#-----------------------------------------------------------------------------
# 添加接口声明宏
#-----------------------------------------------------------------------------
target_compile_definitions(APIServer PRIVATE "APISERVER_LIB")
list(APPEND _depend_library
    
)
list(APPEND _runtimes_libraries 
	Qt5::Core Qt5::Gui Qt5::Widgets 
	Qt5::Network Qt5::WebSockets  MxFunctions
)




#-----------------------------------------------------------------------------
# 链接依赖库
#-----------------------------------------------------------------------------
target_link_libraries(APIServer PRIVATE
	${_runtimes_libraries}
	${_depend_library}	
)



#-----------------------------------------------------------------------------
# 添加依赖关系
#-----------------------------------------------------------------------------
#add_dependencies(MxAiCommon ${_depend_library})

#-----------------------------------------------------------------------------
# 添加运行时依赖关系
#-----------------------------------------------------------------------------
list(APPEND MxPre_Runtimes_Libraries ${_runtimes_libraries})
set(MxPre_Runtimes_Libraries ${MxPre_Runtimes_Libraries} PARENT_SCOPE)

#-----------------------------------------------------------------------------
# 安装头文件
#-----------------------------------------------------------------------------
install(
        FILES ${_header}
        DESTINATION ${INSTALL_INCDIR}                        #将头文件安装到install/include目录
)
